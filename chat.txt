El problema era este flujo:

El Excel de Caldas tiene una celda de fecha vacía → pandas la lee como pd.NaT
sd(pd.NaT) → el check isinstance(NaT, float) da False, así que pasa al try
pd.to_datetime(NaT).date() → devuelve NaT sin lanzar excepción (comportamiento de pandas)
Django intenta guardar NaT en un DateField → llama is_aware(NaT) → llama NaT.utcoffset() → crash
El fix usa pd.isna(v) directamente (que sí detecta NaT), y además valida el resultado de pd.to_datetime() antes de llamar .date(). Prueba Caldas AUTO de nuevo.

def _desc(codigo_concepto, clasificacion):
    clasi = (clasificacion or "").strip()
    desc = _DESC_CLASI.get((codigo_concepto, clasi))
    if desc:
        return desc
    # intenta sin clasificación (fallback a vacío)
    desc = _DESC_CLASI.get((codigo_concepto, ""))
    if desc:
        return desc
    return _DESC_FIJA.get(codigo_concepto, codigo_concepto)


def build_rows(proceso, filtros=None):
    """
    Devuelve (HEADERS, rows) con el formato SAIMYR de 19 columnas.
    filtros: dict con claves opcionales: q, fecha_desde, fecha_hasta, estado
    """
    from etl.models import EncabezadoCXC

    filtros = filtros or {}
    qs = EncabezadoCXC.objects.filter(proceso=proceso).prefetch_related("detalles")

    q = filtros.get("q", "")
    if q:
        qs = (qs.filter(consecutivo_cxc__icontains=q) |
              qs.filter(numero_documento__icontains=q) |
              qs.filter(primer_apellido__icontains=q) |
              qs.filter(razon_social__icontains=q))
    if filtros.get("fecha_desde"):
        qs = qs.filter(fecha_cobro__gte=filtros["fecha_desde"])
    if filtros.get("fecha_hasta"):
        qs = qs.filter(fecha_cobro__lte=filtros["fecha_hasta"])
    if filtros.get("estado"):
        qs = qs.filter(estado_cxc__iexact=filtros["estado"])

    rows = []
    for enc in qs:
        extra = enc.datos_extra or {}
        fecha = enc.fecha_cobro.strftime("%Y-%m-%d") if enc.fecha_cobro else ""
        for det in enc.detalles.all():
            clasi = (det.centro_costo or "").strip()
            rows.append([
                enc.consecutivo_cxc,                    # radicado_externo
                enc.consecutivo_cxc,                    # comprobante_pago
                fecha,                                  # Fecha_pago
                extra.get("tipo_persona", ""),          # Tipo Persona
                enc.tipo_documento,                     # tipo_documento_id
                enc.numero_documento,                   # DI_Tercero
                "",                                     # dv (no almacenado)
                enc.primer_apellido,                    # Priemr_apelledo_tercero
                enc.segundo_apellido,                   # Segundo_apelledo_tercero
                enc.primer_nombre,                      # nombres_tercero
                extra.get("dir_tercero", ""),           # dir_tercero
                extra.get("tel_tercero", ""),           # tel_tercero
                extra.get("email_tercero", ""),         # email_tercero
                clasi,                                  # clasificacion
                _desc(det.codigo_concepto, clasi),      # concepto_externo
                det.codigo_concepto,                    # código_cpto_saimyr
                det.valor_total,                        # valor_pagado
                extra.get("codigo_banco", "1"),         # código_banco
                extra.get("numero_cuenta", "2"),        # número_cuenta
            ])
    return HEADERS, rows

import django, os
os.environ.setdefault('DJANGO_SETTINGS_MODULE', 'core.settings')
django.setup()

import pandas as pd
from muni.models import CIIUMunicipio, Municipio

muni = Municipio.objects.get(codigo='COPACABANA')

df = pd.read_excel('MUNICIPIOS/COPACABANA/INSUMOS/FIJOS/CODIFICACION CIIU.xlsx', dtype=str)
# Columnas: CIIU 4, Descripción, Tarifa, TIPO

# TIPO en el archivo: COMERCIAL, SERVICIOS, INDUSTRIAL
# Normalizar a los valores del modelo
TIPO_MAP = {
    'COMERCIAL': 'COMERCIAL',
    'SERVICIOS': 'SERVICIOS',
    'INDUSTRIAL': 'INDUSTRIAL',
}

creados = 0
for _, row in df.iterrows():
    codigo = str(row.get('CIIU 4', '')).strip().zfill(4)
    tipo = str(row.get('TIPO', '')).strip().upper()
    desc = str(row.get('Descripción', row.get('Descripci\xf3n', ''))).strip()
    if not codigo or codigo == 'nan' or tipo not in TIPO_MAP:
        continue
    obj, created = CIIUMunicipio.objects.update_or_create(
        municipio=muni,
        codigo=codigo,
        defaults={'tipo': TIPO_MAP[tipo], 'descripcion': desc[:300]}
    )
    if created:
        creados += 1

total = CIIUMunicipio.objects.filter(municipio=muni).count()
print(f'Creados: {creados} | Total Copacabana en BD: {total}')
